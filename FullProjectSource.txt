import SwiftUI
import UserNotifications

// 1. Define a custom notification name
extension Notification.Name {
    static let openTransactionTemplate = Notification.Name("openTransactionTemplate")
}

@main
struct BudgetApp: App {
    @StateObject private var data = AppData()
    @UIApplicationDelegateAdaptor(AppDelegate.self) var appDelegate
    
    // Settings
    @AppStorage("isAppLockEnabled") private var isAppLockEnabled: Bool = false
    @AppStorage("selectedTheme") private var selectedTheme: Int = 0
    
    // Lifecycle
    @Environment(\.scenePhase) private var scenePhase
    @State private var isLocked = false
    
    // Hoisted states for shortcut commands
    @State private var selectedTab: Int = 0
    @State private var showAddAccount = false
    @State private var showAddGroup = false
    @State private var showAddTransaction = false

    var body: some Scene {
        WindowGroup {
            ZStack {
                ContentView(
                    selectedTab: $selectedTab,
                    showAddAccount: $showAddAccount,
                    showAddGroup: $showAddGroup,
                    showAddTransaction: $showAddTransaction
                )
                .environmentObject(data)
                .preferredColorScheme(selectedTheme == 0 ? nil : (selectedTheme == 1 ? .light : .dark))
                
                if isLocked {
                    LockView(isLocked: $isLocked)
                        .zIndex(1)
                        .transition(.opacity)
                }
            }
            .onChange(of: scenePhase) { newPhase in
                if newPhase == .inactive || newPhase == .background {
                    if isAppLockEnabled { isLocked = true }
                }
            }
            .onAppear {
                if isAppLockEnabled { isLocked = true }
                
                // CRITICAL FIX: Pass the AppData instance to AppDelegate
                // This ensures templates are loaded before trying to open them
                appDelegate.appData = data
                
                // Process any pending notification that arrived before AppData loaded
                appDelegate.processPendingNotification()
            }
        }
        .commands {
            CommandMenu("Pocket Ledger") {
                Button("Go to Accounts") {
                    selectedTab = 0
                }
                .keyboardShortcut("1", modifiers: [.command])
                
                Button("Go to Groups") {
                    selectedTab = 1
                }
                .keyboardShortcut("2", modifiers: [.command])
                
                Button("Go to Transactions") {
                    selectedTab = 2
                }
                .keyboardShortcut("3", modifiers: [.command])
                
                Divider()
                
                Button("Add Account") {
                    showAddAccount = true
                }
                .keyboardShortcut("A", modifiers: [.command])
                
                Button("Add Group") {
                    showAddGroup = true
                }
                .keyboardShortcut("G", modifiers: [.command])
                
                Button("Add Transaction") {
                    showAddTransaction = true
                }
                .keyboardShortcut("T", modifiers: [.command])
            }
        }
    }
}

// 2. Update Delegate to handle notification taps
class AppDelegate: NSObject, UIApplicationDelegate, UNUserNotificationCenterDelegate {
    
    // NEW: Hold reference to AppData
    weak var appData: AppData?
    
    // NEW: Store pending template ID if notification arrives before AppData is ready
    private var pendingTemplateID: String?
    
    func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]? = nil
    ) -> Bool {
        // Set the delegate so we can capture taps
        let center = UNUserNotificationCenter.current()
        center.delegate = self
        
        center.requestAuthorization(options: [.alert, .sound, .badge]) { _, error in
            if let error = error {
                print("Notification permission error: \(error.localizedDescription)")
            }
        }
        return true
    }

    // This function runs when a user taps a notification
    func userNotificationCenter(_ center: UNUserNotificationCenter,
                                didReceive response: UNNotificationResponse,
                                withCompletionHandler completionHandler: @escaping () -> Void) {
        
        // The Identifier matches the Template UUID string we set in AppData
        let templateID = response.notification.request.identifier
        
        // NEW: Check if AppData is loaded and has templates
        if let appData = appData, !appData.templates.isEmpty {
            // AppData is ready, post immediately
            NotificationCenter.default.post(
                name: .openTransactionTemplate,
                object: nil,
                userInfo: ["id": templateID]
            )
        } else {
            // AppData not ready yet, store for later
            print("AppData not ready, storing pending template: \(templateID)")
            pendingTemplateID = templateID
        }
        
        completionHandler()
    }
    
    // NEW: Process pending notification after AppData loads
    func processPendingNotification() {
        guard let templateID = pendingTemplateID else { return }
        
        // Verify template exists before posting
        if let appData = appData,
           appData.templates.contains(where: { $0.id.uuidString == templateID }) {
            print("Processing pending template: \(templateID)")
            
            // Small delay to ensure UI is ready
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                NotificationCenter.default.post(
                    name: .openTransactionTemplate,
                    object: nil,
                    userInfo: ["id": templateID]
                )
            }
        } else {
            print("Template not found: \(templateID)")
        }
        
        // Clear pending
        pendingTemplateID = nil
    }
    
    // Optional: Allow notifications to show even when app is in foreground
    func userNotificationCenter(_ center: UNUserNotificationCenter,
                                willPresent notification: UNNotification,
                                withCompletionHandler completionHandler: @escaping (UNNotificationPresentationOptions) -> Void) {
        completionHandler([.banner, .sound])
    }
}

import SwiftUI

struct ContentView: View {

            .tabItem { Label("Transact", systemImage: "arrow.left.arrow.right") }
            .tag(1)
            
            // Tab 3: Templates
            TemplatesView()
                .tabItem { Label("Templates", systemImage: "doc.on.doc.fill") }
                .tag(2)
            
            // Tab 4: Insights
            InsightsView()
                .tabItem { Label("Insights", systemImage: "chart.pie.fill") }
                .tag(3)
            
            // Tab 5: More
            SettingsView()
                .tabItem { Label("More", systemImage: "ellipsis.circle.fill") }
                .tag(4)
        }
        .preferredColorScheme(selectedTheme == 0 ? nil : (selectedTheme == 1 ? .light : .dark))
        .onAppear {
            // Sync with default tab setting on launch
            // We only set this if we are at the default 0 state
            if selectedTab == 0 {
                selectedTab = defaultTab
            }
        }
        
        // MARK: - Notification Handling
        .onReceive(NotificationCenter.default.publisher(for: .openTransactionTemplate)) { notification in
            if let idString = notification.userInfo?["id"] as? String,
               let uuid = UUID(uuidString: idString),
               let foundTemplate = data.templates.first(where: { $0.id == uuid }) {
                pendingTemplate = foundTemplate
            }
        }
        .sheet(item: $pendingTemplate) { template in
            AddTransactionView(initialType: template.type, template: template)
                .environmentObject(data)
        }
    }
}

import Foundation
import Combine
import UserNotifications

// MARK: - Data Models

struct AccountStats {
    var totalInflow: Double
    var totalOutflow: Double
    var dailyInflow: Double
    var weeklyInflow: Double
    var monthlyInflow: Double
    var yearlyInflow: Double
    var dailyOutflow: Double
    var weeklyOutflow: Double
    var monthlyOutflow: Double
    var yearlyOutflow: Double
}

// MARK: - AppData Store

/// Central store for all user data in the budgeting app.
@MainActor
class AppData: ObservableObject {
    
    // MARK: - Published Properties (Current Book Data)
    @Published var accounts: [Account] = []
    @Published var accountGroups: [AccountGroup] = []
    @Published var categories: [Category] = []
    @Published var tags: [Tag] = []
    @Published var transactions: [Transaction] = []
    @Published var templates: [TransactionTemplate] = []
    @Published var reminderTime: DateComponents = DateComponents(hour: 21, minute: 0)
    @Published var insightPresets: [InsightPreset] = []

    // MARK: - Book Management
    @Published var availableBooks: [BookMetadata] = []
    @Published var activeBook: BookMetadata?

    // MARK: - File Paths
    
    private var indexURL: URL {
        FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("book_index.json")
    }
    
    private func dataURL(for book: BookMetadata) -> URL {
        FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent(book.fileName)
    }

    init() {
        Task {
            await loadIndex()
        }
    }
    
    // MARK: - Index Operations
    
    private func saveIndex() {
        let index = BookIndex(books: availableBooks, lastActiveBookID: activeBook?.id)
        
        Task.detached(priority: .background) {
            let url = await self.indexURL
            do {
                let data = try JSONEncoder().encode(index)
                try data.write(to: url, options: [.atomic])
            } catch {
                print("Failed to save book index: \(error.localizedDescription)")
            }
        }
    }
    
    private func loadIndex() async {
        let url = indexURL
        
        let loadedIndex = await Task.detached(priority: .userInitiated) { () -> BookIndex? in
            guard FileManager.default.fileExists(atPath: url.path) else { return nil }
            do {
                let data = try Data(contentsOf: url)
                return try JSONDecoder().decode(BookIndex.self, from: data)
            } catch {
                return nil
            }
        }.value
        
        if let index = loadedIndex {
            self.availableBooks = index.books
            
            if let lastID = index.lastActiveBookID,
               let match = index.books.first(where: { $0.id == lastID }) {
                await loadBook(match)
            } else if let firstBook = index.books.first {
                await loadBook(firstBook)
            } else {
                await createDefaultBook()
            }
        } else {
            await createDefaultBook()
        }
    }
    
    private func createDefaultBook() async {
        let defaultBook = BookMetadata(name: "My Ledger")
        self.availableBooks = [defaultBook]
        saveIndex()
        await loadBook(defaultBook)
    }

    // MARK: - Book Loading & Saving

    func loadBook(_ book: BookMetadata) async {
        self.activeBook = book
        
        if let index = availableBooks.firstIndex(where: { $0.id == book.id }) {
            availableBooks[index].lastOpened = Date()
            saveIndex()
        }
        
        let url = dataURL(for: book)
        
        let loadedData = await Task.detached(priority: .userInitiated) { () -> SavedData? in
            do {
                guard FileManager.default.fileExists(atPath: url.path) else { return nil }
                let data = try Data(contentsOf: url)
                return try JSONDecoder().decode(SavedData.self, from: data)
            } catch {
                print("Error decoding book data: \(error.localizedDescription)")
                return nil
            }
        }.value
        
        if let decoded = loadedData {
            self.accounts = decoded.accounts
            self.accountGroups = decoded.accountGroups ?? []
            self.categories = decoded.categories
            self.tags = decoded.tags
            self.transactions = decoded.transactions
            self.templates = decoded.templates ?? []
            self.insightPresets = decoded.insightPresets ?? []
            self.reminderTime = decoded.reminderTime ?? DateComponents(hour: 21, minute: 0)
            updateReminders()
        } else {
            resetToEmpty()
            await initializeDefaultsIfNeeded()
        }
    }

    func save() {
        guard let currentBook = activeBook else { return }
        let url = dataURL(for: currentBook)
        
        let dataToSave = SavedData(
            accounts: accounts,
            accountGroups: accountGroups,
            categories: categories,
            tags: tags,
            transactions: transactions,
            templates: templates,
            reminderTime: reminderTime,
            insightPresets: insightPresets
        )
        
        Task.detached(priority: .background) {
            do {
                let data = try JSONEncoder().encode(dataToSave)
                try data.write(to: url, options: [.atomic, .completeFileProtection])
            } catch {
                print("Error saving data: \(error.localizedDescription)")
            }
        }
    }
    
    // MARK: - CREATE Operations

    func addAccount(name: String,
                    initialAmount: Double,
                    isCredit: Bool = false,
                    targetAmount: Double? = nil,
                    targetDate: Date? = nil,
                    paymentFrequency: PaymentFrequency? = nil) {
        let account = Account(
            name: name,
            initialAmount: initialAmount,
            isCredit: isCredit,
            targetAmount: targetAmount,
            targetDate: targetDate,
            paymentFrequency: paymentFrequency
        )
        accounts.append(account)
        save()
    }
    
    func addAccountGroup(name: String, accountIDs: [UUID]) {
        let group = AccountGroup(name: name, accountIDs: accountIDs)
        accountGroups.append(group)
        save()
    }

    func addCategory(name: String, type: CategoryType, emoji: String) {
        let category = Category(name: name, type: type, emoji: emoji)
        categories.append(category)
        save()
    }

    func addTag(name: String) {
        let tag = Tag(name: name)
        tags.append(tag)
        save()
    }

    func addTransaction(type: TransactionType,
                        accountID: UUID,
                        categoryID: UUID,
                        tagIDs: [UUID],
                        amount: Double,
                        date: Date,
                        isOutlier: Bool = false) {
        let transaction = Transaction(type: type,
                                      accountID: accountID,
                                      categoryID: categoryID,
                                      tagIDs: tagIDs,
                                      amount: amount,
                                      date: date,
                                      isOutlier: isOutlier)
        transactions.append(transaction)
        
        // Auto-Unarchive: If we use an account, it must be active.
        unarchiveAccountIfNeeded(id: accountID)
        
        save()
    }

    // MARK: - UPDATE & DELETE Operations

    func updateAccount(id: UUID,
                       name: String,
                       initialAmount: Double,
                       targetAmount: Double? = nil,
                       targetDate: Date? = nil,
                       paymentFrequency: PaymentFrequency? = nil) {
        if let index = accounts.firstIndex(where: { $0.id == id }) {
            accounts[index].name = name
            accounts[index].initialAmount = initialAmount
            accounts[index].targetAmount = targetAmount
            accounts[index].targetDate = targetDate
            accounts[index].paymentFrequency = paymentFrequency
            save()
        }
    }

    func deleteAccount(_ account: Account) {
        // 1. Remove from physical list
        accounts.removeAll { $0.id == account.id }
        
        // 2. Scrub reference from virtual groups
        for i in 0..<accountGroups.count {
            accountGroups[i].accountIDs.removeAll { $0 == account.id }
        }
        
        save()
    }
    
    func updateAccountGroup(_ group: AccountGroup) {
        if let index = accountGroups.firstIndex(where: { $0.id == group.id }) {
            accountGroups[index] = group
            save()
        }
    }
    
    func deleteAccountGroup(_ group: AccountGroup) {
        if let index = accountGroups.firstIndex(where: { $0.id == group.id }) {
            accountGroups.remove(at: index)
            save()
        }
    }

    func updateCategory(id: UUID, name: String, type: CategoryType, emoji: String) {
        if let index = categories.firstIndex(where: { $0.id == id }) {
            categories[index].name = name
            categories[index].type = type
            categories[index].emoji = emoji
            save()
        }
    }

    func updateTag(id: UUID, name: String) {
        if let index = tags.firstIndex(where: { $0.id == id }) {
            tags[index].name = name
            save()
        }
    }
    
    func updateTransaction(_ transaction: Transaction) {
        if let index = transactions.firstIndex(where: { $0.id == transaction.id }) {
            transactions[index] = transaction
            
            // Auto-Unarchive: Check the account in case it was changed to an archived one
            unarchiveAccountIfNeeded(id: transaction.accountID)
            
            save()
        }
    }

    func removeTagFromTransactions(tagID: UUID) {
        for i in 0..<transactions.count {
            if transactions[i].tagIDs.contains(tagID) {
                transactions[i].tagIDs.removeAll { $0 == tagID }
            }
        }
        save()
    }
    
    // MARK: - Archive Operations (Public)

    func archiveAccount(_ account: Account) {
        if let index = accounts.firstIndex(where: { $0.id == account.id }) {
            accounts[index].archivedDate = Date()
            save()
        }
    }

    func unarchiveAccount(_ account: Account) {
        if let index = accounts.firstIndex(where: { $0.id == account.id }) {
            accounts[index].archivedDate = nil
            save()
        }
    }
    
    /// Helper to automatically unarchive an account if it is used in a new transaction
    private func unarchiveAccountIfNeeded(id: UUID) {
        if let index = accounts.firstIndex(where: { $0.id == id }) {
            if accounts[index].isArchived {
                accounts[index].archivedDate = nil
                // No save() here; it relies on the caller (addTransaction/updateTransaction) to save.
            }
        }
    }

    // MARK: - Template Operations

    func addTemplate(name: String,
                     type: TransactionType,
                     accountID: UUID? = nil,
                     categoryID: UUID? = nil,
                     tagIDs: [UUID] = [],
                     isOutlier: Bool = false,
                     transferTargetAccountID: UUID? = nil) {
        let template = TransactionTemplate(name: name,
                                           type: type,
                                           accountID: accountID,
                                           categoryID: categoryID,
                                           tagIDs: tagIDs,
                                           isOutlier: isOutlier,
                                           transferTargetAccountID: transferTargetAccountID)
        templates.append(template)
        save()
    }
    
    func updateTemplate(_ template: TransactionTemplate) {
        if let index = templates.firstIndex(where: { $0.id == template.id }) {
            templates[index] = template
            save()
            updateReminders()
        }
    }

    func removeTemplate(_ template: TransactionTemplate) {
        templates.removeAll { $0.id == template.id }
        save()
        updateReminders()
    }
    
    // MARK: - Preset Operations
    
    func savePreset(_ preset: InsightPreset) {
        if let index = insightPresets.firstIndex(where: { $0.id == preset.id }) {
            insightPresets[index] = preset
        } else {
            insightPresets.append(preset)
        }
        save()
    }
    
    func deletePreset(_ preset: InsightPreset) {
        insightPresets.removeAll { $0.id == preset.id }
        save()
    }

    // MARK: - Reminder Logic

    func updateReminders() {
        let center = UNUserNotificationCenter.current()
        center.removeAllPendingNotificationRequests()
        
        for template in templates {
            if let schedule = template.reminderSchedule {
                scheduleNotification(for: template, schedule: schedule)
            }
        }
    }

    private func scheduleNotification(for template: TransactionTemplate, schedule: ReminderSchedule) {
        let center = UNUserNotificationCenter.current()
        
        let content = UNMutableNotificationContent()
        content.title = "Record Transaction"
        content.body = "Time to record: \(template.name)"
        content.sound = .default
        
        var dateComponents = DateComponents()
        dateComponents.hour = schedule.time.hour
        dateComponents.minute = schedule.time.minute
        
        switch schedule.frequency {
        case .daily:
            break
        case .weekly:
            dateComponents.weekday = schedule.dayOfWeek
        case .monthly:
            dateComponents.day = schedule.dayOfMonth
        }
        
        let trigger = UNCalendarNotificationTrigger(dateMatching: dateComponents, repeats: true)
        let request = UNNotificationRequest(identifier: template.id.uuidString, content: content, trigger: trigger)
        
        center.add(request) { error in
            if let error = error {
                print("Error scheduling notification: \(error.localizedDescription)")
            }
        }
    }

    // MARK: - Data Management

    func resetAllData() {
        resetToEmpty()
        save()
        Task { await initializeDefaultsIfNeeded() }
    }
    
    private func resetToEmpty() {
        accounts = []
        accountGroups = []
        categories = []
        tags = []
        transactions = []
        templates = []
        insightPresets = []
        UNUserNotificationCenter.current().removeAllPendingNotificationRequests()
    }
    
    func restoreFromBackup(url: URL) throws {
        guard url.startAccessingSecurityScopedResource() else {
            throw NSError(domain: "Permissions", code: 403, userInfo: [NSLocalizedDescriptionKey: "Permission denied to access file."])
        }
        defer { url.stopAccessingSecurityScopedResource() }
        
        let data = try Data(contentsOf: url)
        let decoded = try JSONDecoder().decode(SavedData.self, from: data)
        
        self.accounts = decoded.accounts
        self.accountGroups = decoded.accountGroups ?? []
        self.categories = decoded.categories
        self.tags = decoded.tags
        self.transactions = decoded.transactions
        self.templates = decoded.templates ?? []
        self.insightPresets = decoded.insightPresets ?? []
        self.reminderTime = decoded.reminderTime ?? DateComponents(hour: 21, minute: 0)
        
        save()
        updateReminders()
    }
    
    @MainActor
    func initializeDefaultsIfNeeded() async {
        if accounts.isEmpty {
            addAccount(name: "Cash", initialAmount: 0.0)
            addAccount(name: "Bank", initialAmount: 0.0)
        }
        
        if categories.isEmpty {
            addCategory(name: "Food", type: .outflow, emoji: "🍽")
            addCategory(name: "Housing", type: .outflow, emoji: "🏠")
            addCategory(name: "Transportation", type: .outflow, emoji: "🚗")
            addCategory(name: "Utilities", type: .outflow, emoji: "💡")
            addCategory(name: "Entertainment", type: .outflow, emoji: "🎮")
            addCategory(name: "Shopping", type: .outflow, emoji: "🛍")
            addCategory(name: "Health", type: .outflow, emoji: "💊")
            addCategory(name: "Gifts", type: .outflow, emoji: "🎁")
            addCategory(name: "Other", type: .outflow, emoji: "📦")
            addCategory(name: "Salary", type: .inflow, emoji: "💼")
            addCategory(name: "Bonus", type: .inflow, emoji: "💰")
            addCategory(name: "Other Income", type: .inflow, emoji: "📥")
        }
        
        if tags.isEmpty {
            addTag(name: "Mandatory")
            addTag(name: "Discretionary")
        }
    }
    
    // MARK: - Query Helpers

    func balance(for account: Account) -> Double {
        let inflow = transactions
            .filter { $0.accountID == account.id && $0.type == .inflow }
            .reduce(0) { $0 + $1.amount }
        let outflow = transactions
            .filter { $0.accountID == account.id && $0.type == .outflow }
            .reduce(0) { $0 + $1.amount }
        
        if account.isCredit {
            return account.initialAmount + outflow - inflow
        } else {
            return account.initialAmount + inflow - outflow
        }
    }
    
    func balance(for group: AccountGroup) -> Double {
        group.accountIDs.reduce(0) { partialResult, accountID in
            if let account = accounts.first(where: { $0.id == accountID }) {
                let accountBalance = balance(for: account)
                if account.isCredit {
                    return partialResult - accountBalance
                } else {
                    return partialResult + accountBalance
                }
            }
            return partialResult
        }
    }

    func transactions(for account: Account) -> [Transaction] {
        transactions.filter { $0.accountID == account.id }
    }

    func transactions(for category: Category) -> [Transaction] {
        transactions.filter { $0.categoryID == category.id }
    }

    func transactions(for tag: Tag) -> [Transaction] {
        transactions.filter { $0.tagIDs.contains(tag.id) }
    }

    func stats(for account: Account) -> AccountStats {
        let accountTransactions = transactions(for: account)
        let inflowTransactions = accountTransactions.filter { $0.type == .inflow }
        let outflowTransactions = accountTransactions.filter { $0.type == .outflow }

        let totalInflow = inflowTransactions.reduce(0) { $0 + $1.amount }
        let totalOutflow = outflowTransactions.reduce(0) { $0 + $1.amount }

        let dates = accountTransactions.map { $0.date }
        let start = dates.min() ?? account.created
        let end = dates.max() ?? account.created
        
        let dayComponents = Calendar.current.dateComponents([.day], from: start, to: end)
        let totalDays = max((dayComponents.day ?? 0) + 1, 1)
        
        let weeks = max(Double(totalDays) / 7.0, 1.0)
        let months = max(Double(totalDays) / 30.0, 1.0)
        let years = max(Double(totalDays) / 365.0, 1.0)

        return AccountStats(
            totalInflow: totalInflow,
            totalOutflow: totalOutflow,
            dailyInflow: totalInflow / Double(totalDays),
            weeklyInflow: totalInflow / weeks,
            monthlyInflow: totalInflow / months,
            yearlyInflow: totalInflow / years,
            dailyOutflow: totalOutflow / Double(totalDays),
            weeklyOutflow: totalOutflow / weeks,
            monthlyOutflow: totalOutflow / months,
            yearlyOutflow: totalOutflow / years
        )
    }
}

import Foundation

/// Represents the lightweight header information for a specific book.
/// This allows the app to list books without loading their full transaction history.
struct BookMetadata: Identifiable, Codable, Hashable, Sendable {
    let id: UUID
    var name: String
    var created: Date
    var lastOpened: Date
    
    /// Generates a unique filename for this specific book's data.
    var fileName: String {
        "book_\(id.uuidString).json"
    }
    
    init(id: UUID = UUID(), name: String, created: Date = Date(), lastOpened: Date = Date()) {
        self.id = id
        self.name = name
        self.created = created
        self.lastOpened = lastOpened
    }
}

/// The master record stored in 'book_index.json'.
/// It tracks which books exist and which one was used last.
struct BookIndex: Codable, Sendable {
    var books: [BookMetadata]
    var lastActiveBookID: UUID?
}

/// Structure for persisting all user data to disk.
struct SavedData: Codable, Sendable {
    var accounts: [Account]
    var accountGroups: [AccountGroup]?
    var categories: [Category]
    var tags: [Tag]
    var transactions: [Transaction]
    var templates: [TransactionTemplate]?
    var reminderTime: DateComponents?
    var insightPresets: [InsightPreset]?
    
    // MARK: - Migration Support
    
    /// Custom decoder to handle migration from old Account structure
    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        
        // Decode all fields normally
        self.accountGroups = try container.decodeIfPresent([AccountGroup].self, forKey: .accountGroups)
        self.categories = try container.decode([Category].self, forKey: .categories)
        self.tags = try container.decode([Tag].self, forKey: .tags)
        self.transactions = try container.decode([Transaction].self, forKey: .transactions)
        self.templates = try container.decodeIfPresent([TransactionTemplate].self, forKey: .templates)
        self.reminderTime = try container.decodeIfPresent(DateComponents.self, forKey: .reminderTime)
        self.insightPresets = try container.decodeIfPresent([InsightPreset].self, forKey: .insightPresets)
        
        // Decode accounts with migration logic
        var decodedAccounts = try container.decode([Account].self, forKey: .accounts)
        
        // MIGRATION: Add default values for new fields if they're missing
        for i in 0..<decodedAccounts.count {
            // If account is credit and has no targetAmount, it's a Debt account
            if decodedAccounts[i].isCredit && decodedAccounts[i].targetAmount == nil {
                decodedAccounts[i].targetAmount = nil // Explicitly keep as nil for debt tracking
            }
            
            // targetDate and paymentFrequency will naturally be nil if not present in old data
        }
        
        self.accounts = decodedAccounts
    }
    
    // MARK: - Standard Initializer
    
    init(accounts: [Account],
         accountGroups: [AccountGroup]? = nil,
         categories: [Category],
         tags: [Tag],
         transactions: [Transaction],
         templates: [TransactionTemplate]? = nil,
         reminderTime: DateComponents? = nil,
         insightPresets: [InsightPreset]? = nil) {
        self.accounts = accounts
        self.accountGroups = accountGroups
        self.categories = categories
        self.tags = tags
        self.transactions = transactions
        self.templates = templates
        self.reminderTime = reminderTime
        self.insightPresets = insightPresets
    }
}

import Foundation

enum TransactionType: String, Codable, CaseIterable, Identifiable, Sendable {
    case inflow
    case outflow
    case transfer
    var id: String { rawValue }
}

struct Transaction: Identifiable, Codable, Hashable, Sendable {
    let id: String
    var type: TransactionType
    var accountID: UUID
    var categoryID: UUID
    var tagIDs: [UUID]
    var amount: Double
    var date: Date
    var isOutlier: Bool = false

    init(
        id: String? = nil,
        type: TransactionType,
        accountID: UUID,
        categoryID: UUID,
        tagIDs: [UUID] = [],
        amount: Double,
        date: Date = Date(),
        isOutlier: Bool = false
    ) {
        self.id = id ?? Transaction.generateUniqueTimestampID()
        self.type = type
        self.accountID = accountID
        self.categoryID = categoryID
        self.tagIDs = tagIDs
        self.amount = amount
        self.date = date
        self.isOutlier = isOutlier
    }

    // MARK: - Safe ID Generation
    private static func generateUniqueTimestampID() -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyyMMddHHmmSSS"
        
        // 1. Generate Timestamp
        let timestamp = formatter.string(from: Date())
        
        // 2. Generate 3 random digits (000-999) to prevent collisions
        // This replaces the unsafe 'static counter'
        let random = Int.random(in: 0...999)
        let randomSuffix = String(format: "%03d", random)
        
        return timestamp + randomSuffix
    }
}

import Foundation
import SwiftUI

/// Represents a financial container such as a bank account, credit card, savings goal, or debt.
///
/// Accounts can be categorized into four types based on `isCredit` and `targetAmount`:
/// - **Debit** (isCredit: false, targetAmount: nil): Cash, checking accounts
/// - **Goal** (isCredit: false, targetAmount: set): Savings goals with target dates
/// - **Budget** (isCredit: true, targetAmount: set): Spending envelopes with monthly limits
/// - **Debt** (isCredit: true, targetAmount: nil or 0): Loans, installments with payoff goals
struct Account: Identifiable, Codable, Hashable, Sendable {
    /// Unique identifier for the account.
    let id: UUID
    /// Human-readable name for the account.
    var name: String
    /// Starting balance when the account is first recorded.
    var initialAmount: Double
    /// Date when the account record was created.
    var created: Date
    
    /// Indicates whether this account functions as a credit account.
    /// - For debit accounts: balance increases with inflows, decreases with outflows
    /// - For credit accounts: balance increases with outflows (debt/spending), decreases with inflows (payments)
    var isCredit: Bool = false
    
    /// Optional target amount for goal tracking, budget limits, or debt payoff.
    /// - For debit accounts: represents a savings goal to reach
    /// - For budget accounts: represents a spending limit
    /// - For debt accounts: typically nil or 0 (payoff goal)
    var targetAmount: Double? = nil
    
    /// Optional target date for reaching a goal or paying off debt.
    /// Used to calculate required periodic contributions/payments.
    var targetDate: Date? = nil
    
    /// Frequency of contributions (Goals) or payments (Debts).
    /// Used with targetDate to calculate installment amounts.
    var paymentFrequency: PaymentFrequency? = nil
    
    /// Date when the account was archived. If nil, account is active.
    var archivedDate: Date? = nil
    
    /// Computed property to check if account is archived
    var isArchived: Bool {
        archivedDate != nil
    }
    
    init(id: UUID = UUID(),
         name: String,
         initialAmount: Double = 0.0,
         created: Date = Date(),
         isCredit: Bool = false,
         targetAmount: Double? = nil,
         targetDate: Date? = nil,
         paymentFrequency: PaymentFrequency? = nil,
         archivedDate: Date? = nil) {
        self.id = id
        self.name = name
        self.initialAmount = initialAmount
        self.created = created
        self.isCredit = isCredit
        self.targetAmount = targetAmount
        self.targetDate = targetDate
        self.paymentFrequency = paymentFrequency
        self.archivedDate = archivedDate
    }
    
    /// Computed property for account type classification
    var accountType: AccountType {
        switch (isCredit, targetAmount) {
        case (false, nil):
            return .debit
        case (false, .some(let target)) where target > 0:
            return .goal
        case (true, .some(let target)) where target > 0:
            return .budget
        case (true, nil), (true, .some(0)):
            return .debt
        default:
            return .debt
        }
    }
}

// MARK: - Supporting Enums

enum AccountType: String, Codable {
    case debit = "Debit"
    case goal = "Goal"
    case budget = "Budget"
    case debt = "Debt"
    
    /// SF Symbol icon for each account type
    var icon: String {
        switch self {
        case .debit:  return "banknote"
        case .goal:   return "target"
        case .budget: return "envelope.fill"
        case .debt:   return "creditcard.fill"
        }
    }
    
    /// SwiftUI Color for each account type
    var color: Color {
        switch self {
        case .debit:  return .green
        case .goal:   return .blue
        case .budget: return .purple
        case .debt:   return .orange
        }
    }
    
    /// Section header name (for grouping)
    var sectionName: String {
        self.rawValue
    }
}

enum PaymentFrequency: String, Codable, CaseIterable, Identifiable {
    case weekly = "Weekly"
    case biweekly = "Bi-weekly"
    case monthly = "Monthly"
    case quarterly = "Quarterly"
    case yearly = "Yearly"
    
    var id: String { rawValue }
    
    /// Returns the number of periods per year
    var periodsPerYear: Double {
        switch self {
        case .weekly:    return 52
        case .biweekly:  return 26
        case .monthly:   return 12
        case .quarterly: return 4
        case .yearly:    return 1
        }
    }
    
    /// Calculates total number of periods between two dates
    func periodsBetween(from start: Date, to end: Date) -> Int {
        let calendar = Calendar.current
        let components: Calendar.Component
        let multiplier: Int
        
        switch self {
        case .weekly:
            components = .weekOfYear
            multiplier = 1
        case .biweekly:
            components = .weekOfYear
            multiplier = 2
        case .monthly:
            components = .month
            multiplier = 1
        case .quarterly:
            components = .month
            multiplier = 3
        case .yearly:
            components = .year
            multiplier = 1
        }
        
        let difference = calendar.dateComponents([components], from: start, to: end)
        let rawPeriods = difference.value(for: components) ?? 0
        
        if multiplier == 1 {
            return rawPeriods
        } else {
            return rawPeriods / multiplier
        }
    }
}

import SwiftUI

struct ManageBooksView: View {
    @EnvironmentObject var data: AppData
    @Environment(\.dismiss) var dismiss
    
    // UI State for Adding
    @State private var showingAddAlert = false
    @State private var newBookName = ""
    
    // UI State for Deleting
    @State private var bookToDelete: BookMetadata?
    @State private var showingDeleteAlert = false
    
    // UI State for Renaming (NEW)
    @State private var bookToEdit: BookMetadata?
    @State private var showingEditAlert = false
    @State private var editBookName = ""

    var body: some View {
        List {
            Section(footer: Text("Tap a book to switch to it. Swipe right to rename.")) {
                ForEach(data.availableBooks) { book in
                    Button {
                        switchToBook(book)
                    } label: {
                        HStack {
                            VStack(alignment: .leading) {
                                Text(book.name)
                                    .font(.headline)
                                    .foregroundColor(.primary)
                                Text("Last opened: \(book.lastOpened.formatted(date: .abbreviated, time: .shortened))")
                                    .font(.caption)
                                    .foregroundColor(.secondary)
                            }
                            
                            Spacer()
                            
                            // Checkmark for active book
                            if data.activeBook?.id == book.id {
                                Image(systemName: "checkmark.circle.fill")
                                    .foregroundColor(.blue)
                            }
                        }
                    }
                    // SWIPE RIGHT (Leading): Rename
                    .swipeActions(edge: .leading) {
                        Button {
                            prepareRename(book)
                        } label: {
                            Label("Rename", systemImage: "pencil")
                        }
                        .tint(.blue)
                    }
                    // SWIPE LEFT (Trailing): Delete
                    .swipeActions(edge: .trailing) {
                        if data.activeBook?.id != book.id {
                            Button(role: .destructive) {
                                bookToDelete = book
                                showingDeleteAlert = true
                            } label: {
                                Label("Delete", systemImage: "trash")
                            }
                        }
                    }
                }
            }
        }
        .navigationTitle("My Books")
        .toolbar {
            ToolbarItem(placement: .primaryAction) {
                Button {
                    newBookName = ""
                    showingAddAlert = true
                } label: {
                    Image(systemName: "plus")
                }
            }
        }
        
        // ALERT: Add Book
        .alert("New Book", isPresented: $showingAddAlert) {
            TextField("Book Name", text: $newBookName)
            Button("Cancel", role: .cancel) { }
            Button("Create") {
                if !newBookName.isEmpty {
                    createNewBook(name: newBookName)
                }
            }
        }
        
        // ALERT: Rename Book (NEW)
        .alert("Rename Book", isPresented: $showingEditAlert) {
            TextField("New Name", text: $editBookName)
            Button("Cancel", role: .cancel) { }
            Button("Save") {
                if let book = bookToEdit, !editBookName.isEmpty {
                    renameBook(book, to: editBookName)
                }
            }
        }
        
        // ALERT: Delete Confirmation
        .alert("Delete Book?", isPresented: $showingDeleteAlert, presenting: bookToDelete) { book in
            Button("Cancel", role: .cancel) { }
            Button("Delete", role: .destructive) {
                deleteBook(book)
            }
        } message: { book in
            Text("Are you sure you want to delete '\(book.name)'? This will permanently erase all transactions in this book.")
        }
    }
    
    // MARK: - Logic
    
    private func switchToBook(_ book: BookMetadata) {
        guard data.activeBook?.id != book.id else { return }
        Task {
            await data.loadBook(book)
        }
    }
    
    private func createNewBook(name: String) {
        let newBook = BookMetadata(name: name)
        data.availableBooks.append(newBook)
        Task {
            await data.loadBook(newBook)
        }
    }
    
    // Logic for Renaming
    private func prepareRename(_ book: BookMetadata) {
        bookToEdit = book
        editBookName = book.name
        showingEditAlert = true
    }
    
    private func renameBook(_ book: BookMetadata, to newName: String) {
        // 1. Find and update in the list
        if let index = data.availableBooks.firstIndex(where: { $0.id == book.id }) {
            data.availableBooks[index].name = newName
            
            // 2. If it's the active book, update that reference too so the UI refreshes instantly
            if data.activeBook?.id == book.id {
                data.activeBook?.name = newName
            }
            
            // 3. Trigger persistence
            if let current = data.activeBook {
                Task { await data.loadBook(current) }
            }
        }
    }
    
    private func deleteBook(_ book: BookMetadata) {
        data.availableBooks.removeAll { $0.id == book.id }
        
        let fileURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent(book.fileName)
        
        try? FileManager.default.removeItem(at: fileURL)
        
        if let current = data.activeBook {
            Task { await data.loadBook(current) }
        }
    }
}

    /// Processes the result of a JSON backup import.

        switch result {
        case .success(let urls):
            if let url = urls.first {
                do {
                    try data.restoreFromBackup(url: url)
                    triggerHaptic(.success)
                } catch {
                    importError = "Could not restore backup. Please ensure the file is a valid Pocket Ledger JSON file."
                    showingImportAlert = true
                    triggerHaptic(.error)
                }
            }
        case .failure(let error):
            importError = error.localizedDescription
            showingImportAlert = true
            triggerHaptic(.error)
        }
    }

    /// Requests FaceID/TouchID permissions for the app lock feature.
    private func requestBiometricPermission() {
        let context = LAContext()
        var error: NSError?
        if context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) {
            context.evaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, localizedReason: "Lock Pocket Ledger") { success, _ in
                DispatchQueue.main.async {
                    if !success { isAppLockEnabled = false }
                }
            }
        } else {
            isAppLockEnabled = false
        }
    }
}

                .foregroundColor(.blue)
            
            Text("Pocket Ledger is Locked")
                .font(.title2)
                .fontWeight(.bold)
            
            if !errorText.isEmpty {
                Text(errorText)
                    .foregroundColor(.red)
                    .font(.caption)
                    .multilineTextAlignment(.center)
                    .padding(.horizontal)
            }
            
            Button("Unlock") {
                authenticate()
            }
            .buttonStyle(.borderedProminent)
            .padding(.top)
        }
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .background(Color(UIColor.systemBackground))
        .ignoresSafeArea()
        // Use .task instead of .onAppear for better async handling
        .task {
            // Wait slightly for the app to finish the "opening" animation
            try? await Task.sleep(nanoseconds: 500_000_000) // 0.5 seconds
            authenticate()
        }
    }

    private func authenticate() {
        let context = LAContext()
        var error: NSError?

        // Check if device supports FaceID/TouchID
        if context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) {
            let reason = "Unlock your ledger"

            context.evaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, localizedReason: reason) { success, authenticationError in
                DispatchQueue.main.async {
                    if success {
                        // Success: Unlock
                        isLocked = false
                        errorText = ""
                    } else {
                        // Failure: Show specific error
                        if let err = authenticationError {
                            // If it was canceled by system, tell user to tap button
                            if let laError = err as? LAError, laError.code == .systemCancel {
                                errorText = "System interrupted Face ID. Tap Unlock to try again."
                            } else if let laError = err as? LAError, laError.code == .userCancel {
                                errorText = "Authentication canceled."
                            } else {
                                errorText = err.localizedDescription
                            }
                        } else {
                            errorText = "Authentication failed."
                        }
                    }
                }
            }
        } else {
            // Fallback for devices without FaceID or if permissions are missing
            errorText = "Biometrics not available. Please ensure Face ID is enabled for Pocket Ledger in iOS Settings."
        }
    }
}

    @State private var actualBalanceInputs: [UUID: String] = [:]
    
    // State for the adjustment sheet (replaced alert with sheet for more detail)
    @State private var accountToAdjust: Account?
    @State private var difference: Double = 0
    @State private var showingAdjustmentSheet = false
    
    // Filter state
    @State private var showOnlyWithDifferences = false
    
    // MARK: - Focus State
    @FocusState private var focusedAccountID: UUID?
    
    // MARK: - Filtered Accounts
    
    private var displayedAccounts: [Account] {
        let accounts = data.accounts.sorted { lhs, rhs in
            // Sort by account type, then alphabetically
            let typeOrder: [AccountType] = [.debit, .goal, .budget, .debt]
            let lhsIndex = typeOrder.firstIndex(of: lhs.accountType) ?? 999
            let rhsIndex = typeOrder.firstIndex(of: rhs.accountType) ?? 999
            
            if lhsIndex != rhsIndex {
                return lhsIndex < rhsIndex
            }
            return lhs.name.localizedCaseInsensitiveCompare(rhs.name) == .orderedAscending
        }
        
        if showOnlyWithDifferences {
            return accounts.filter { account in
                if let diff = calculateDifference(for: account), abs(diff) > 0.01 {
                    return true
                }
                return false
            }
        }
        
        return accounts
    }
    
    // MARK: - Grouped Accounts
    
    private var accountsByType: [(type: AccountType, accounts: [Account])] {
        let typeOrder: [AccountType] = [.debit, .goal, .budget, .debt]
        
        return typeOrder.compactMap { type in
            let accountsOfType = displayedAccounts.filter { $0.accountType == type }
            return accountsOfType.isEmpty ? nil : (type, accountsOfType)
        }
    }

                            .font(.caption)
                            .foregroundColor(.secondary)

                }

                if differenceSummary.count > 0 || differenceSummary.total != 0 {

                        HStack {
                            VStack(alignment: .leading) {
                                Text("Accounts with differences")
                                    .font(.caption)
                                    .foregroundColor(.secondary)
                                Text("\(differenceSummary.count)")
                                    .font(.title3)
                                    .fontWeight(.bold)
                            }
                            
                            Spacer()
                            
                            VStack(alignment: .trailing) {
                                Text("Total difference")
                                    .font(.caption)
                                    .foregroundColor(.secondary)
                                Text(differenceSummary.total.formattedCurrency(symbol: currencySymbol))
                                    .font(.title3)
                                    .fontWeight(.bold)
                                    .foregroundColor(differenceSummary.total > 0 ? .green : .red)
                            }

        return actual - system

        var count = 0
        var total: Double = 0

                count += 1
                total += diff

    private func fixAllDifferences() {

        generator.notificationOccurred(.success)
    }
    
    private func createAdjustmentFor(account: Account, difference: Double, categoryID: UUID?) {
        let type: TransactionType = difference > 0 ? .inflow : .outflow
        let amount = abs(difference)
        
        // Determine category
        let finalCategoryID: UUID
        
        if let providedID = categoryID {
            finalCategoryID = providedID
        } else {
            // Auto-select category
            let validCategories = data.categories.filter { $0.type.rawValue == type.rawValue }
            
            var targetCategory = validCategories.first { $0.name == "Other" }
            if targetCategory == nil {
                targetCategory = validCategories.first { $0.name == "Adjustment" }
            }
            if targetCategory == nil {
                targetCategory = validCategories.first
            }
            
            guard let category = targetCategory else { return }
            finalCategoryID = category.id
        }
        
        data.addTransaction(
            type: type,
            accountID: account.id,
            categoryID: finalCategoryID,
            tagIDs: [],
            amount: amount,
            date: Date(),
            isOutlier: true // Mark reconciliation adjustments as outliers by default
        )

            minAmountString = "100"

// MARK: - CSV Document
struct CSVDocument: FileDocument {
    static var readableContentTypes: [UTType] { [.commaSeparatedText] }

    var text: String

    init(text: String) {
        self.text = text
    }

    init(configuration: ReadConfiguration) throws {
        self.text = ""
    }

    func fileWrapper(configuration: WriteConfiguration) throws -> FileWrapper {
        let data = Data(text.utf8)
        return FileWrapper(regularFileWithContents: data)
    }
}

// MARK: - JSON Document
struct JSONBackupDocument: FileDocument {
    static var readableContentTypes: [UTType] { [.json] }
    
    var savedData: SavedData
    
    init(data: SavedData) {
        self.savedData = data
    }
    
    init(configuration: ReadConfiguration) throws {
        // FIX: Removed 'try' because regularFileContents does not throw
        let data = configuration.file.regularFileContents ?? Data()
        self.savedData = try JSONDecoder().decode(SavedData.self, from: data)
    }
    
    func fileWrapper(configuration: WriteConfiguration) throws -> FileWrapper {
        let encoder = JSONEncoder()
        encoder.outputFormatting = .prettyPrinted
        let data = try encoder.encode(savedData)
        return FileWrapper(regularFileWithContents: data)
    }
}

struct ExportView: View {

    // Exporter State
    @State private var showCSVExporter = false
    @State private var showJSONExporter = false
    
    @State private var csvDocument: CSVDocument?
    @State private var jsonDocument: JSONBackupDocument?

            } else {
                Section(footer: Text("Creates a full backup file of all accounts, transactions, and settings.")) {

                }
            }
            
            Section {

                    HStack {
                        Spacer()
                        Text(exportFormat == .csv ? "Export CSV" : "Export Backup")
                            .fontWeight(.semibold)

                    }
                }
                // Only disable for CSV if date range is invalid
                .disabled(exportFormat == .csv && endDate < startDate)
            }
        }
        .navigationTitle("Export Data")
        // CSV Exporter
        .fileExporter(
            isPresented: $showCSVExporter,
            document: csvDocument,
            contentType: .commaSeparatedText,
            defaultFilename: defaultCSVName()
        ) { result in
            handleResult(result)
        }
        // JSON Exporter
        .fileExporter(
            isPresented: $showJSONExporter,
            document: jsonDocument,
            contentType: .json,
            defaultFilename: "PocketLedger_Backup_\(Date().formatted(date: .numeric, time: .omitted))"
        ) { result in
            handleResult(result)
        }
    }
    
    // MARK: - Logic
    
    private func handleExport() {
        if exportFormat == .csv {
            csvDocument = prepareCSVDocument()
            showCSVExporter = true
        } else {
            jsonDocument = prepareJSONDocument()
            showJSONExporter = true
        }
    }
    
    private func handleResult(_ result: Result<URL, Error>) {
        switch result {
        case .success(let url):
            print("Exported to \(url)")
            dismiss()
        case .failure(let error):
            print("Export failed: \(error.localizedDescription)")
        }
    }

        let snapshot = SavedData(
            accounts: data.accounts,
            categories: data.categories,
            tags: data.tags,
            transactions: data.transactions,
            templates: data.templates,
            reminderTime: data.reminderTime,
            insightPresets: data.insightPresets
        )
        return JSONBackupDocument(data: snapshot)
    }

    private func prepareCSVDocument() -> CSVDocument {
        let filtered = data.transactions
            .filter { $0.date >= startDate && $0.date <= endDate }
            .sorted(by: { $0.date < $1.date })

        var csv = "ID,Date,Type,Account,Category,Tags,Amount\n"
        let dateFormatter = DateFormatter()
        dateFormatter.dateFormat = "yyyy-MM-dd"
        
        for tx in filtered {
            let accountName = data.accounts.first(where: { $0.id == tx.accountID })?.name ?? "Unknown"
            let categoryName = data.categories.first(where: { $0.id == tx.categoryID })?.name ?? "Unknown"
            let tagsString = tx.tagIDs
                .compactMap { id in data.tags.first(where: { $0.id == id })?.name }
                .joined(separator: "; ")
            
            let dateStr = dateFormatter.string(from: tx.date)
            let amountStr = String(format: "%.2f", tx.amount)
            
            // Clean strings to prevent CSV breakage
            let safeTags = tagsString.replacingOccurrences(of: ",", with: " ")
            
            csv += "\(tx.id),\(dateStr),\(tx.type.rawValue),\(accountName),\(categoryName),\(safeTags),\(amountStr)\n"
        }
        return CSVDocument(text: csv)
    }

    private func defaultCSVName() -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyyMMdd"
        return "Transactions_\(formatter.string(from: startDate))_\(formatter.string(from: endDate))"
    }
}

    struct AnalyticsStats {
        var income: Double
        var expense: Double
        var net: Double { income - expense }
        var breakdownItems: [AnalyticsItem]

    struct AnalyticsItem: Identifiable {
        let id: UUID
        let name: String
        let icon: String // Emoji or SF Symbol name
        let amount: Double
        let color: Color? // Optional override
    }
    
    private func calculateStats(for range: TimeRange, offset: Int) -> AnalyticsStats {
        let dateInterval = range.dateInterval(relativeTo: selectedAnchorDate, fiscalStartDay: fiscalStartDay, offset: offset)
        
        // 1. Identify "Transfer" categories to exclude from Income/Expense
        let transferCategoryIDs = Set(data.categories.filter { $0.name == "Transfer" }.map { $0.id })
        
        // 2. Filter Transactions
        let filtered = data.transactions.filter { tx in
            // Time Filter
            if !dateInterval.contains(tx.date) { return false }
            
            // Standard Filters
            if excludeOutliers && tx.isOutlier { return false }
            if excludedAccountIDs.contains(tx.accountID) { return false }
            if excludedCategoryIDs.contains(tx.categoryID) { return false }
            if !tx.tagIDs.isEmpty && !Set(tx.tagIDs).isDisjoint(with: excludedTagIDs) { return false }
            
            // Amount Filter
            if let min = minimumAmount, tx.amount < min { return false }
            if let max = maximumAmount, tx.amount > max { return false }
            
            return true
        }
        
        // 3. Calculate Totals (excluding transfers)
        let income = filtered
            .filter { $0.type == .inflow && !transferCategoryIDs.contains($0.categoryID) }
            .reduce(0) { $0 + $1.amount }
        
        let expense = filtered
            .filter { $0.type == .outflow && !transferCategoryIDs.contains($0.categoryID) }
            .reduce(0) { $0 + $1.amount }
        
        // 4. Generate Breakdown (only for expenses usually, but can adapt)
        let outflows = filtered.filter { $0.type == .outflow && !transferCategoryIDs.contains($0.categoryID) }
        var breakdown: [AnalyticsItem] = []
        
        switch breakdownMode {
        case .category:
            var dict: [UUID: Double] = [:]
            for tx in outflows { dict[tx.categoryID, default: 0] += tx.amount }
            breakdown = dict.compactMap { (id, amount) in
                guard let cat = data.categories.first(where: { $0.id == id }) else { return nil }
                return AnalyticsItem(id: id, name: cat.name, icon: cat.emoji, amount: amount, color: nil)
            }
            
        case .account:
            var dict: [UUID: Double] = [:]
            for tx in outflows { dict[tx.accountID, default: 0] += tx.amount }
            breakdown = dict.compactMap { (id, amount) in
                guard let acc = data.accounts.first(where: { $0.id == id }) else { return nil }
                return AnalyticsItem(id: id, name: acc.name, icon: acc.accountType.icon, amount: amount, color: acc.accountType.color)
            }
            
        case .tag:
            // Tag logic: If a transaction has 2 tags, add amount to BOTH.
            var dict: [UUID: Double] = [:]
            var untaggedAmount: Double = 0
            
            for tx in outflows {
                if tx.tagIDs.isEmpty {
                    untaggedAmount += tx.amount
                } else {
                    for tagID in tx.tagIDs {
                        dict[tagID, default: 0] += tx.amount
                    }
                }
            }
            breakdown = dict.compactMap { (id, amount) in
                guard let tag = data.tags.first(where: { $0.id == id }) else { return nil }
                return AnalyticsItem(id: id, name: tag.name, icon: "#", amount: amount, color: .blue)
            }
            if untaggedAmount > 0 {
                breakdown.append(AnalyticsItem(id: UUID(), name: "No Tag", icon: "tag.slash", amount: untaggedAmount, color: .secondary))
            }
        }
        
        return AnalyticsStats(
            income: income,
            expense: expense,
            breakdownItems: breakdown.sorted { $0.amount > $1.amount }
        )
    }

// MARK: - Enums & Components

enum TimeRange: String, CaseIterable {
    case thisMonth = "This Month"
    case lastMonth = "Last Month"
    case last3Months = "Last 3 Months"
    case ytd = "YTD"
    case lastYear = "Last Year"
    case allTime = "All Time"
    
    func dateInterval(relativeTo anchor: Date, fiscalStartDay: Int, offset: Int = 0) -> DateInterval {
        let calendar = Calendar.current
        var components = calendar.dateComponents([.year, .month], from: anchor)
        components.day = fiscalStartDay
        
        // Calculate the "Current Month Start" based on fiscal day
        var currentMonthStart = calendar.date(from: components)!
        if anchor < currentMonthStart {
            currentMonthStart = calendar.date(byAdding: .month, value: -1, to: currentMonthStart)!
        }
        
        switch self {
        case .thisMonth:
            let start = calendar.date(byAdding: .month, value: offset, to: currentMonthStart)!
            let end = calendar.date(byAdding: .month, value: 1, to: start)!
            return DateInterval(start: start, end: end)
            
        case .lastMonth:
            let baseStart = calendar.date(byAdding: .month, value: -1, to: currentMonthStart)!
            let start = calendar.date(byAdding: .month, value: offset, to: baseStart)!
            let end = calendar.date(byAdding: .month, value: 1, to: start)!
            return DateInterval(start: start, end: end)
            
        case .last3Months:
            let durationMonths = 3
            let shift = offset * durationMonths
            let currentMonthEnd = calendar.date(byAdding: .month, value: 1, to: currentMonthStart)!
            let windowEnd = calendar.date(byAdding: .month, value: shift, to: currentMonthEnd)!
            let windowStart = calendar.date(byAdding: .month, value: -durationMonths, to: windowEnd)!
            return DateInterval(start: windowStart, end: windowEnd)
            
        case .ytd:
            let currentYearStart = calendar.date(from: DateComponents(year: calendar.component(.year, from: anchor), month: 1, day: 1))!
            let yearShift = offset
            let start = calendar.date(byAdding: .year, value: yearShift, to: currentYearStart)!
            let end = calendar.date(byAdding: .year, value: yearShift, to: Date())!
            return DateInterval(start: start, end: end)
            
        case .lastYear:
            let currentYear = calendar.component(.year, from: anchor)
            let targetYear = currentYear - 1 + offset
            let start = calendar.date(from: DateComponents(year: targetYear, month: 1, day: 1))!
            let end = calendar.date(from: DateComponents(year: targetYear + 1, month: 1, day: 1))!
            return DateInterval(start: start, end: end)
            
        case .allTime:
            return DateInterval(start: Date.distantPast, end: Date.distantFuture)
        }
    }
}

import SwiftUI
import Foundation

struct CurrencyFormatter {
    /// format: 0.00
    static func string(from value: Double) -> String {
        return String(format: "%.2f", value)
    }
    
    /// format: $0.00 (or user selected symbol)
    static func display(for value: Double, with symbol: String) -> String {
        return "\(symbol)\(string(from: value))"
    }
}

// Optional: A reusable ViewModifier if you want to inject the symbol easily
struct CurrencyDisplay: View {
    @AppStorage("currencySymbol") private var currencySymbol: String = "$"
    let amount: Double
    
    var body: some View {
        Text("\(currencySymbol)\(String(format: "%.2f", amount))")
    }
}

extension Double {
    /// Formats the double as a currency string with thousand separators (e.g., "$1,234.56")
    func formattedCurrency(symbol: String) -> String {
        let formatter = NumberFormatter()
        formatter.numberStyle = .currency
        formatter.currencySymbol = symbol
        formatter.usesGroupingSeparator = true // Enables "1,000"
        formatter.minimumFractionDigits = 2
        formatter.maximumFractionDigits = 2
        
        return formatter.string(from: NSNumber(value: self)) ?? "\(symbol)0.00"
    }
}

/// A generic searchable list for single selection (Accounts/Categories) with visual enhancements
struct SelectionView<T: Identifiable>: View where T: Hashable {
    @EnvironmentObject var data: AppData
    @AppStorage("currencySymbol") private var currencySymbol: String = "$"
    
    let title: String
    let items: [T]
    @Binding var selectedID: T.ID?
    @Environment(\.dismiss) var dismiss
    @State private var searchText = ""
    
    @FocusState private var isSearchFocused: Bool
    
    // NEW: State for collapsed archived section
    @State private var isArchivedExpanded: Bool = false
    
    var filteredItems: [T] {
        let list: [T]
        if searchText.isEmpty {
            list = items
        } else {
            list = items.filter { item in
                if let named = item as? Account {
                    return named.name.localizedCaseInsensitiveContains(searchText)
                }
                if let named = item as? Category {
                    return named.name.localizedCaseInsensitiveContains(searchText)
                }
                return true
            }
        }
        
        return list.sorted { first, second in
            var name1 = ""
            var name2 = ""
            if let a = first as? Account { name1 = a.name }
            else if let c = first as? Category { name1 = c.name }
            if let a = second as? Account { name2 = a.name }
            else if let c = second as? Category { name2 = c.name }
            return name1.localizedCaseInsensitiveCompare(name2) == .orderedAscending
        }
    }
    
    private var isAccountView: Bool {
        T.self == Account.self
    }
    
    // MARK: - Account Grouping Logic
    
    // Active accounts grouped by type
    private var activeAccountsByType: [(type: AccountType, accounts: [T])] {
        guard isAccountView else { return [] }
        
        let typeOrder: [AccountType] = [.debit, .goal, .budget, .debt]
        // Filter only Active accounts
        let activeItems = filteredItems.filter { !($0 as? Account)!.isArchived }
        
        return typeOrder.compactMap { type in
            let accountsOfType = activeItems.filter {
                ($0 as? Account)?.accountType == type
            }
            return accountsOfType.isEmpty ? nil : (type, accountsOfType)
        }
    }
    
    // Archived accounts list
    private var archivedAccounts: [T] {
        guard isAccountView else { return [] }
        return filteredItems.filter { ($0 as? Account)!.isArchived }
    }
    
    var body: some View {
        VStack(spacing: 0) {
            searchHeader
            
            List {
                if isAccountView {
                    // 1. Active Accounts (Grouped by Type)
                    ForEach(activeAccountsByType, id: \.type) { section in
                        Section(header: sectionHeader(for: section.type)) {
                            ForEach(section.accounts) { item in
                                accountSelectionRow(for: item)
                            }
                        }
                    }
                    
                    // 2. Archived Accounts (Collapsed Section)
                    if !archivedAccounts.isEmpty {
                        DisclosureGroup(isExpanded: $isArchivedExpanded) {
                            ForEach(archivedAccounts) { item in
                                accountSelectionRow(for: item)
                            }
                        } label: {
                            HStack {
                                Image(systemName: "archivebox")
                                Text("Archived Accounts")
                                Spacer()
                                Text("\(archivedAccounts.count)")
                                    .foregroundColor(.secondary)
                                    .font(.caption)
                            }
                            .foregroundColor(.secondary)
                        }
                    }
                    
                } else {
                    // CATEGORIES: Standard list with emoji
                    ForEach(filteredItems) { item in
                        categorySelectionRow(for: item)
                    }
                }
            }
        }
        .navigationTitle(title)
        .onAppear {
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
                isSearchFocused = true
            }
        }
    }
    
    // MARK: - Section Header
    
    @ViewBuilder

    private func balanceColor(for account: Account) -> Color {
        if account.isArchived { return .secondary }
        
        let balance = data.balance(for: account)
        
        switch account.accountType {
        case .debit:
            return balance < 0 ? .red : .secondary
        case .goal:
            if let target = account.targetAmount, balance >= target {
                return .green
            }
            return .secondary
        case .budget:
            if let target = account.targetAmount, balance > target {
                return .red
            }
            return .secondary
        case .debt:
            return balance > 0 ? .orange : .green
        }
    }

/// A searchable list for multiple selection (Tags) with Keyboard-first creation logic
struct MultiSelectionView: View {
    @EnvironmentObject var data: AppData
    let title: String
    let items: [Tag]
    @Binding var selectedIDs: Set<UUID>
    @Environment(\.dismiss) var dismiss
    
    @State private var searchText = ""
    @FocusState private var isSearchFocused: Bool
    
    var filteredItems: [Tag] {
        let list = searchText.isEmpty
            ? items
            : items.filter { $0.name.localizedCaseInsensitiveContains(searchText) }
        return list.sorted { $0.name.localizedCaseInsensitiveCompare($1.name) == .orderedAscending }
    }
    
    var body: some View {
        VStack(spacing: 0) {
            searchHeader
            
            List {
                if showCreationRow {
                    Button(action: createAndSelectTag) {
                        HStack {
                            Image(systemName: "plus.circle.fill").foregroundColor(.green)
                            Text("Create \"\(searchText)\"")
                                .fontWeight(.medium)
                        }
                    }
                }

                ForEach(filteredItems) { item in
                    Button {
                        toggleSelection(item)
                    } label: {
                        HStack {
                            Text(item.name)
                            Spacer()
                            if selectedIDs.contains(item.id) {
                                Image(systemName: "checkmark").foregroundColor(.blue)
                            }
                        }
                    }
                    .foregroundColor(.primary)
                }
            }
        }
        .navigationTitle(title)
        .onAppear {
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
                isSearchFocused = true
            }
        }
    }
    
    private var searchHeader: some View {
        HStack {
            Image(systemName: "tag")
                .foregroundColor(.secondary)
            TextField("Find or create tags...", text: $searchText)
                .focused($isSearchFocused)
                .submitLabel(.done)
                .onSubmit {
                    handleKeyboardSubmit()
                }
            if !searchText.isEmpty {
                Button { searchText = "" } label: {
                    Image(systemName: "xmark.circle.fill").foregroundColor(.secondary)
                }
            }
        }
        .padding(8)
        .background(Color(.secondarySystemBackground))
        .cornerRadius(10)
        .padding()
    }
    
    private var showCreationRow: Bool {
        !searchText.isEmpty && !items.contains(where: { $0.name.caseInsensitiveCompare(searchText) == .orderedSame })
    }
    
    private func handleKeyboardSubmit() {
        if searchText.isEmpty {
            dismiss()
        } else if filteredItems.count == 1 {
            toggleSelection(filteredItems[0])
            searchText = ""
            isSearchFocused = true
        } else if showCreationRow {
            createAndSelectTag()
        }
    }
    
    private func toggleSelection(_ item: Tag) {
        if selectedIDs.contains(item.id) {
            selectedIDs.remove(item.id)
        } else {
            selectedIDs.insert(item.id)
        }
    }
    
    private func createAndSelectTag() {
        let cleanName = searchText.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !cleanName.isEmpty else { return }
        
        data.addTag(name: cleanName)
        
        if let newTag = data.tags.first(where: { $0.name == cleanName }) {
            selectedIDs.insert(newTag.id)
        }
        
        searchText = ""
        isSearchFocused = true
    }
}

        if shouldDismiss {

        }
    }

import SwiftUI

/// Displays all transactions tagged with a specific tag, optionally filtered by a date range.
struct TagDetailView: View {
    @EnvironmentObject var data: AppData
    @AppStorage("currencySymbol") private var currencySymbol: String = "$"
    
    var tag: Tag
    // NEW: Optional filter passed from Insights
    var dateFilter: DateInterval? = nil

    var body: some View {
        List {
            // NEW: Summary Card if filtered
            if let interval = dateFilter {
                Section {
                    VStack(alignment: .leading, spacing: 4) {
                        Text("Total for Period")
                            .font(.caption)
                            .foregroundColor(.secondary)
                        
                        Text(totalAmount.formattedCurrency(symbol: currencySymbol))
                            .font(.title2)
                            .fontWeight(.bold)
                            .foregroundColor(.primary)
                        
                        Text(formatInterval(interval))
                            .font(.caption2)
                            .foregroundColor(.secondary)
                    }
                    .padding(.vertical, 4)
                }
            }

            Section(header: Text("Transactions")) {
                if filteredTransactions.isEmpty {
                    Text("No transactions found for this period.")
                        .foregroundColor(.secondary)
                } else {
                    ForEach(filteredTransactions) { transaction in
                        // Assuming you have TransactionDetailView available
                        NavigationLink(destination: TransactionDetailView(transaction: transaction)) {
                            TransactionRow(transaction: transaction)
                        }
                    }
                }
            }
        }
        .navigationTitle("#\(tag.name)")
    }
    
    // MARK: - Helpers
    
    private var filteredTransactions: [Transaction] {
        let all = data.transactions(for: tag)
        guard let interval = dateFilter else {
            return all.sorted { $0.date > $1.date }
        }
        return all
            .filter { interval.contains($0.date) }
            .sorted { $0.date > $1.date }
    }
    
    private var totalAmount: Double {
        filteredTransactions.reduce(0) { $0 + $1.amount }
    }
    
    private func formatInterval(_ interval: DateInterval) -> String {
        let formatter = DateFormatter()
        formatter.dateStyle = .medium
        formatter.timeStyle = .none
        return "\(formatter.string(from: interval.start)) - \(formatter.string(from: interval.end))"
    }
}

    private func calculateAmount(_ input: String) -> Double {
        let allowed = CharacterSet(charactersIn: "0123456789.+-*/")
        let sanitized = input.components(separatedBy: allowed.inverted).joined()
        if sanitized.isEmpty { return 0 }
        guard let lastChar = sanitized.last, lastChar.isNumber else {
            return Double(sanitized) ?? 0
        }
        let expression = NSExpression(format: sanitized)
        if let result = expression.expressionValue(with: nil, context: nil) as? Double {
            return result
        } else if let intResult = expression.expressionValue(with: nil, context: nil) as? Int {
            return Double(intResult)
        }
        return Double(sanitized) ?? 0

        case .account:
            if accountID == nil {
                showAccountPicker = true

            fallthrough // Proceed to check next
            
        case .secondary:
            if type == .transfer {

                    showTransferTargetPicker = true
                    return

            } else {
                if categoryID == nil {
                    showCategoryPicker = true
                    return
                }
            }
            fallthrough
            
        case .tags:
            if selectedTagIDs.isEmpty {
                showTagPicker = true
                return
            }
            // If tags are filled, we stop (flow complete, user can review/save)
        }
    }

        }
    }
    
    private func advanceAfterTransferTargetSelection() {

            checkAndPresentNextField(startingFrom: .tags)
        }
    }
    
    private func advanceAfterCategorySelection() {
        guard categoryID != nil else { return }
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.3) {
            checkAndPresentNextField(startingFrom: .tags)
        }
    }
    
    private func saveAndAddAnother() {
        saveTransaction(shouldDismiss: false)
        amountString = ""
        selectedTagIDs = []
        isOutlier = false
        focusedField = .amount
        let generator = UINotificationFeedbackGenerator()
        generator.notificationOccurred(.success)
    }

    private func saveTransaction(shouldDismiss: Bool = true) {
        let amount = calculateAmount(amountString)
        let tags = Array(selectedTagIDs)

        if type == .transfer {
            guard let fromAcc = accountID, let toAcc = transferTargetAccountID else { return }
            
            let outCatID = ensureTransferCategory(type: .outflow)
            let inCatID = ensureTransferCategory(type: .inflow)
            
            data.addTransaction(type: .outflow, accountID: fromAcc, categoryID: outCatID, tagIDs: tags, amount: amount, date: date, isOutlier: isOutlier)
            data.addTransaction(type: .inflow, accountID: toAcc, categoryID: inCatID, tagIDs: tags, amount: amount, date: date, isOutlier: isOutlier)
            
        } else {
            guard let accountID = accountID, let categoryID = categoryID else { return }
            
            if let existing = editingTransaction {
                var updated = existing
                updated.type = type
                updated.accountID = accountID
                updated.categoryID = categoryID
                updated.tagIDs = tags
                updated.amount = amount
                updated.date = date
                updated.isOutlier = isOutlier
                data.updateTransaction(updated)
            } else {
                if installmentCount <= 1 {
                    data.addTransaction(type: type, accountID: accountID, categoryID: categoryID, tagIDs: tags, amount: amount, date: date, isOutlier: isOutlier)
                } else {
                    let perInstallment = amount / Double(installmentCount)
                    for i in 0..<installmentCount {
                        let instDate = Calendar.current.date(byAdding: .month, value: i, to: date) ?? date
                        data.addTransaction(type: type, accountID: accountID, categoryID: categoryID, tagIDs: tags, amount: perInstallment, date: instDate, isOutlier: isOutlier)
                    }
                }
            }
        }
        
        if shouldDismiss {
            dismiss()
        }
    }
    
    private func ensureTransferCategory(type: CategoryType) -> UUID {
        if let existing = data.categories.first(where: { $0.name == "Transfer" && $0.type == type }) {
            return existing.id
        }
        let newCat = Category(name: "Transfer", type: type, emoji: "⇆")
        data.categories.append(newCat)
        data.save()
        return newCat.id
    }
}

struct BatchTagSelectionContent: View {
    @EnvironmentObject var data: AppData
    @Binding var selectedIDs: Set<UUID>
    let onApply: () -> Void
    
    @State private var searchText = ""
    @FocusState private var isSearchFocused: Bool
    
    var filteredItems: [Tag] {
        let list = searchText.isEmpty
            ? data.tags
            : data.tags.filter { $0.name.localizedCaseInsensitiveContains(searchText) }
        return list.sorted { $0.name.localizedCaseInsensitiveCompare($1.name) == .orderedAscending }
    }
    
    var showCreationRow: Bool {
        !searchText.isEmpty && !data.tags.contains(where: { $0.name.caseInsensitiveCompare(searchText) == .orderedSame })
    }
    
    var body: some View {
        VStack(spacing: 0) {
            // Search Header
            HStack {
                Image(systemName: "tag")
                    .foregroundColor(.secondary)
                TextField("Find or create tags...", text: $searchText)
                    .focused($isSearchFocused)
                    .submitLabel(.done)
                    .onSubmit {
                        handleKeyboardSubmit()
                    }
                if !searchText.isEmpty {
                    Button {
                        searchText = ""
                    } label: {
                        Image(systemName: "xmark.circle.fill")
                            .foregroundColor(.secondary)
                    }
                }
            }
            .padding(8)
            .background(Color(.secondarySystemBackground))
            .cornerRadius(10)
            .padding()
            
            // Tag List
            List {
                if showCreationRow {
                    Button(action: createAndSelectTag) {
                        HStack {
                            Image(systemName: "plus.circle.fill")
                                .foregroundColor(.green)
                            Text("Create \"\(searchText)\"")
                                .fontWeight(.medium)
                        }
                    }
                }

                ForEach(filteredItems) { item in
                    Button {
                        toggleSelection(item)
                    } label: {
                        HStack {
                            Text(item.name)
                            Spacer()
                            if selectedIDs.contains(item.id) {
                                Image(systemName: "checkmark.circle.fill")
                                    .foregroundColor(.blue)
                                    .fontWeight(.bold)
                            } else {
                                Image(systemName: "circle")
                                    .foregroundColor(.secondary.opacity(0.3))
                            }
                        }
                    }
                    .foregroundColor(.primary)
                }
                
                // Spacer for bottom toolbar
                Color.clear.frame(height: 80).listRowBackground(Color.clear)
            }
        }
        .onAppear {
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
                isSearchFocused = true
            }
        }
    }
    
    // MARK: - Helpers
    
    private func handleKeyboardSubmit() {
        if searchText.isEmpty {
            // Empty search + Enter = Apply (if tags selected)
            if !selectedIDs.isEmpty {
                onApply()
            }
        } else if filteredItems.count == 1 {
            // Single result + Enter = Select it and clear search
            toggleSelection(filteredItems[0])
            searchText = ""
            isSearchFocused = true
        } else if showCreationRow {
            // Create row visible + Enter = Create and select
            createAndSelectTag()
        }
    }
    
    private func toggleSelection(_ item: Tag) {
        if selectedIDs.contains(item.id) {
            selectedIDs.remove(item.id)
        } else {
            selectedIDs.insert(item.id)
        }
    }
    
    private func createAndSelectTag() {
        let cleanName = searchText.trimmingCharacters(in: .whitespaces)
        guard !cleanName.isEmpty else { return }
        
        data.addTag(name: cleanName)
        
        if let newTag = data.tags.first(where: { $0.name == cleanName }) {
            selectedIDs.insert(newTag.id)
        }
        
        searchText = ""
        isSearchFocused = true
    }
}

        default:

    // MARK: - Installment Calculation
    
    private func calculateInstallment() -> (periodsRemaining: Int, amountPerPeriod: Double)? {
        guard let targetDate = liveAccount.targetDate,
              let frequency = liveAccount.paymentFrequency else {
            return nil
        }
        
        let today = Date()
        guard targetDate > today else { return nil }
        
        let periodsRemaining = frequency.periodsBetween(from: today, to: targetDate)
        guard periodsRemaining > 0 else { return nil }
        
        let current = data.balance(for: liveAccount)
        let remaining: Double
        
        switch liveAccount.accountType {
        case .goal:
            if let target = liveAccount.targetAmount {
                remaining = max(target - current, 0)
            } else {
                return nil
            }
            
        case .debt:
            remaining = max(current, 0)
            
        default:
            return nil
        }
        
        let amountPerPeriod = remaining / Double(periodsRemaining)
        return (periodsRemaining, amountPerPeriod)
    }

            let rangeText = formatFiscalRange(range)
            Text("Current period: \(rangeText)")
        } else {
            EmptyView()
        }
    }

    private func formatFiscalRange(_ range: (start: Date, end: Date)) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "MMM d, yyyy"
        return "\(formatter.string(from: range.start)) - \(formatter.string(from: range.end))"
    }
    
    private func calculateCurrentMonthTotals() -> (inflow: Double, outflow: Double) {
        let range = getCurrentFiscalRange()
        let accountTransactions = data.transactions(for: liveAccount)
        
        let thisMonthTransactions = accountTransactions.filter {
            $0.date >= range.start && $0.date <= range.end
        }
        
        let inflow = thisMonthTransactions
            .filter { $0.type == .inflow }
            .reduce(0) { $0 + $1.amount }
            
        let outflow = thisMonthTransactions
            .filter { $0.type == .outflow }
            .reduce(0) { $0 + $1.amount }
            
        return (inflow, outflow)
    }

    private func getCurrentFiscalRange() -> (start: Date, end: Date) {
        let now = Date()
        let calendar = Calendar.current
        let components = calendar.dateComponents([.year, .month], from: now)
        
        var startComponents = DateComponents()
        startComponents.year = components.year
        startComponents.month = components.month
        startComponents.day = fiscalStartDay
        
        var start = calendar.date(from: startComponents) ?? now
        
        if now < start {
            start = calendar.date(byAdding: .month, value: -1, to: start) ?? start
        }
        
        let end = calendar.date(byAdding: .month, value: 1, to: start)?.addingTimeInterval(-1) ?? now
        return (start, end)
    }
}

import SwiftUI

struct AddAccountGroupView: View {
    @EnvironmentObject var data: AppData
    @Environment(\.dismiss) var dismiss
    
    // Optional: If editing, pass the existing group
    let groupToEdit: AccountGroup?
    
    @State private var name: String
    @State private var selectedAccountIDs: Set<UUID>
    @State private var searchText: String = ""
    
    @AppStorage("currencySymbol") private var currencySymbol: String = "$"
    
    // MARK: - Focus State
    enum Field {
        case name
        case search
    }
    @FocusState private var focusedField: Field?
    
    // MARK: - Explicit Initializer
    init(groupToEdit: AccountGroup? = nil) {
        self.groupToEdit = groupToEdit
        
        if let group = groupToEdit {
            _name = State(initialValue: group.name)
            _selectedAccountIDs = State(initialValue: Set(group.accountIDs))
        } else {
            _name = State(initialValue: "")
            _selectedAccountIDs = State(initialValue: [])
        }
    }
    
    // MARK: - Filter Logic
    var filteredAccounts: [Account] {
        let all = data.accounts
        
        let filtered = searchText.isEmpty ? all : all.filter {
            $0.name.localizedCaseInsensitiveContains(searchText)
        }
        
        // Sort by: 1) Account Type, 2) Name (alphabetical)
        return filtered.sorted { lhs, rhs in
            // Primary sort: Account type order (Debit, Goal, Budget, Debt)
            let typeOrder: [AccountType] = [.debit, .goal, .budget, .debt]
            let lhsIndex = typeOrder.firstIndex(of: lhs.accountType) ?? 999
            let rhsIndex = typeOrder.firstIndex(of: rhs.accountType) ?? 999
            
            if lhsIndex != rhsIndex {
                return lhsIndex < rhsIndex
            }
            
            // Secondary sort: Alphabetical by name
            return lhs.name.localizedCaseInsensitiveCompare(rhs.name) == .orderedAscending
        }
    }
    
    // MARK: - Grouped Accounts
    private var accountsByType: [(type: AccountType, accounts: [Account])] {
        let typeOrder: [AccountType] = [.debit, .goal, .budget, .debt]
        
        return typeOrder.compactMap { type in
            let accountsOfType = filteredAccounts.filter { $0.accountType == type }
            return accountsOfType.isEmpty ? nil : (type, accountsOfType)
        }
    }
    
    var body: some View {
        NavigationStack {
            Form {
                // MARK: - Name Section
                Section(header: Text("Group Name")) {
                    TextField("E.g., Liquid Cash, Travel Fund", text: $name)
                        .focused($focusedField, equals: .name)
                        .submitLabel(.next)
                        .onSubmit {
                            focusedField = .search
                        }
                }
                
                // MARK: - Search Section
                Section(header: Text("Include Accounts")) {
                    HStack {
                        Image(systemName: "magnifyingglass")
                            .foregroundColor(.secondary)
                        
                        TextField("Filter accounts...", text: $searchText)
                            .focused($focusedField, equals: .search)
                            .submitLabel(searchText.isEmpty ? .done : (filteredAccounts.count == 1 ? .join : .search))
                            .onSubmit {
                                handleSearchSubmit()
                            }
                        
                        if !searchText.isEmpty {
                            Button {
                                searchText = ""
                            } label: {
                                Image(systemName: "xmark.circle.fill")
                                    .foregroundColor(.secondary)
                            }
                            .buttonStyle(.plain)
                        }
                    }
                }
                
                // MARK: - Account List (Grouped by Type)
                if filteredAccounts.isEmpty {
                    Section {
                        Text("No accounts found")
                            .foregroundColor(.secondary)
                    }
                } else {
                    // Group accounts by type for visual clarity
                    ForEach(accountsByType, id: \.type) { section in
                        Section(header: Text(section.type.rawValue)) {
                            ForEach(section.accounts) { account in
                                Button {
                                    toggleSelection(for: account)
                                } label: {
                                    HStack(spacing: 12) {
                                        AccountTypeIcon(type: account.accountType, size: 32)
                                        
                                        VStack(alignment: .leading, spacing: 4) {
                                            HStack(spacing: 6) {
                                                Text(account.name)
                                                    .foregroundColor(.primary)
                                                
                                                if let badge = account.statusBadge(balance: data.balance(for: account)) {
                                                    AccountStatusBadgeView(badge: badge)
                                                }
                                            }
                                            
                                            Text(account.accountType.rawValue.capitalized)
                                                .font(.caption)
                                                .foregroundColor(.secondary)
                                        }
                                        
                                        Spacer()
                                        
                                        Text(data.balance(for: account).formattedCurrency(symbol: currencySymbol))
                                            .font(.subheadline)
                                            .foregroundColor(balanceColor(for: account))
                                        
                                        if selectedAccountIDs.contains(account.id) {
                                            Image(systemName: "checkmark.circle.fill")
                                                .foregroundColor(.blue)
                                                .fontWeight(.bold)
                                        } else {
                                            Image(systemName: "circle")
                                                .foregroundColor(.secondary.opacity(0.3))
                                        }
                                    }
                                    .contentShape(Rectangle())
                                }
                                .buttonStyle(.plain)
                            }
                        }
                    }
                }
            }
            .formToolbar(
                title: groupToEdit == nil ? "New Group" : "Edit Group",
                canSave: canSave,
                onSave: { saveGroup() },
                onCancel: { dismiss() }
            )
            .onAppear {
                if groupToEdit == nil {
                    focusedField = .name
                }
            }
        }
    }
    
    // MARK: - Helpers
    
    private var canSave: Bool {
        !name.trimmingCharacters(in: .whitespaces).isEmpty && !selectedAccountIDs.isEmpty
    }
    
    private func toggleSelection(for account: Account) {
        if selectedAccountIDs.contains(account.id) {
            selectedAccountIDs.remove(account.id)
        } else {
            selectedAccountIDs.insert(account.id)
        }
    }
    
    private func handleSearchSubmit() {
        if searchText.isEmpty {
            if canSave {
                saveGroup()
            }
        } else if filteredAccounts.count == 1 {
            if let account = filteredAccounts.first {
                toggleSelection(for: account)
                searchText = ""
            }
        }
    }
    
    private func saveGroup() {
        if var group = groupToEdit {
            group.name = name
            group.accountIDs = Array(selectedAccountIDs)
            data.updateAccountGroup(group)
        } else {
            data.addAccountGroup(name: name, accountIDs: Array(selectedAccountIDs))
        }
        dismiss()
    }
    
    // MARK: - Styling Helpers
    
    private func balanceColor(for account: Account) -> Color {
        let balance = data.balance(for: account)
        
        switch account.accountType {
        case .debit:
            return balance < 0 ? .red : .primary
        case .goal:
            if let target = account.targetAmount, balance >= target {
                return .green
            }
            return .primary
        case .budget:
            if let target = account.targetAmount, balance > target {
                return .red
            }
            return .primary
        case .debt:
            return balance > 0 ? .orange : .green
        }
    }

        case (true, nil):    return "creditcard"             // Credit
        case (false, .some): return "target"                 // Goal
        case (true, .some):  return "chart.bar.doc.horizontal" // Budget
        }
    }
    
    private func accountTypeLabel(for account: Account) -> String {
        switch (account.isCredit, account.targetAmount) {
        case (false, nil):   return "Debit"
        case (true, nil):    return "Credit"
        case (false, .some): return "Goal"
        case (true, .some):  return "Budget"
        }
    }
    
    private func accountTypeLabelColor(for account: Account) -> Color {
        switch (account.isCredit, account.targetAmount) {
        case (false, .some): return .blue    // Goal
        case (true, .some):  return .purple  // Budget
        default:             return .secondary
        }
    }
    
    private func accountBalanceColor(for account: Account) -> Color {
        let balance = data.balance(for: account)
        
        if account.isCredit {
            // Credit: Warn if high
            if balance > account.initialAmount * 0.9 {
                return .red
            } else if balance > account.initialAmount * 0.7 {
                return .orange
            }
        } else {
            // Debit: Warn if negative
            if balance < 0 {
                return .red
            }
        }
        return .primary
    }
    
    /// Shows how this account contributes to the group total
    private func contributionText(for account: Account) -> String {
        let balance = data.balance(for: account)
        
        if account.isCredit {
            // Credit accounts are subtracted from total
            return "(\(balance.formattedCurrency(symbol: currencySymbol)))"
        } else {
            // Debit accounts are added to total
            return "+\(balance.formattedCurrency(symbol: currencySymbol))"
        }
    }
    
    /// Color for the total balance
    private var totalBalanceColor: Color {
        let total = data.balance(for: group)
        return total < 0 ? .red : .primary
    }
    
    /// Explains how the balance is calculated
    private var balanceCalculationExplanation: String {
        let hasDebit = group.accountIDs.contains { id in
            data.accounts.first(where: { $0.id == id })?.isCredit == false
        }
        let hasCredit = group.accountIDs.contains { id in
            data.accounts.first(where: { $0.id == id })?.isCredit == true
        }
        
        if hasDebit && hasCredit {
            return "Group total = Debit accounts (+) minus Credit accounts (−). Credit balances shown in parentheses are subtracted from the total."
        } else if hasCredit {
            return "Group total = Sum of all credit balances. Negative total means net debt."
        } else {
            return "Group total = Sum of all debit balances."
        }
    }
    
    // MARK: - Statistics Calculation
    
    private func calculateGroupMonthlyTotals() -> (inflow: Double, outflow: Double) {
        let transactions = data.transactions.filter { group.accountIDs.contains($0.accountID) }
        let range = getCurrentFiscalRange()
        let filtered = transactions.filter { $0.date >= range.start && $0.date <= range.end }
        
        let inflow = filtered.filter { $0.type == .inflow }.reduce(0) { $0 + $1.amount }
        let outflow = filtered.filter { $0.type == .outflow }.reduce(0) { $0 + $1.amount }
        return (inflow, outflow)
    }

    private func getCurrentFiscalRange() -> (start: Date, end: Date) {
        let calendar = Calendar.current
        let components = calendar.dateComponents([.year, .month], from: Date())
        let startComp = DateComponents(year: components.year, month: components.month, day: fiscalStartDay)
        var start = calendar.date(from: startComp) ?? Date()
        if Date() < start { start = calendar.date(byAdding: .month, value: -1, to: start) ?? start }
        let end = calendar.date(byAdding: .month, value: 1, to: start)?.addingTimeInterval(-1) ?? Date()
        return (start, end)
    }
}

                }
            }
            
        case .debt:
            Text(currentBalance.formattedCurrency(symbol: currencySymbol))

            let target = calculateAmount(targetAmountString)
            return target > 0 // Only show if user adds a target
        }
    }
    
    private var installmentSectionHeader: String {
        account.isCredit ? "Payment Plan (Optional)" : "Contribution Plan (Optional)"
    }
    
    private var installmentSectionFooter: String {
        switch account.accountType {
        case .goal:
            return "Set a target date to calculate how much you need to save each period to reach your goal."
        case .debt:
            return "Set a target date to calculate required payments to pay off this debt."
        default:
            return ""
        }
    }
    
    private var currentBalanceColor: Color {
        let balance = data.balance(for: account)
        
        switch account.accountType {
        case .debit:
            return balance < 0 ? .red : .primary
        case .goal:
            if let target = Double(targetAmountString), balance >= target {
                return .green
            }
            return .primary
        case .budget:
            if let target = Double(targetAmountString), balance > target {
                return .red
            }
            return .primary
        case .debt:
            return balance > 0 ? .orange : .green
        }
    }
    
    private var progressText: String {
        let target = calculateAmount(targetAmountString)
        guard target > 0 else { return "" }
        
        let current = data.balance(for: account)
        let percentage = (current / target) * 100
        
        switch account.accountType {
        case .goal:
            if current >= target {
                return "Goal reached! 🎉"
            } else {
                return String(format: "%.0f%% of goal reached", percentage)
            }
        case .budget:
            return String(format: "%.0f%% of budget used", percentage)
        case .debt:
            return String(format: "%.0f%% of debt remaining", percentage)
        default:
            return ""
        }
    }
    
    // MARK: - Validation
    
    private var canSave: Bool {
        !name.trimmingCharacters(in: .whitespaces).isEmpty
    }
    
    // MARK: - Math Engine
    
    private func calculateAmount(_ input: String) -> Double {
        // 1. Filter allowed characters
        let allowed = CharacterSet(charactersIn: "0123456789.+-*/")
        let sanitized = input.components(separatedBy: allowed.inverted).joined()
        
        if sanitized.isEmpty { return 0 }
        
        // 2. If it ends in a math symbol, just parse the safe part to avoid crash
        if let lastChar = sanitized.last, !lastChar.isNumber {
            return Double(sanitized.dropLast()) ?? 0
        }
        
        // 3. Evaluate expression
        let expression = NSExpression(format: sanitized)
        if let result = expression.expressionValue(with: nil, context: nil) as? Double {
            return result
        } else if let intResult = expression.expressionValue(with: nil, context: nil) as? Int {
            return Double(intResult)
        }
        
        return Double(sanitized) ?? 0
    }
    
    // MARK: - Calculations
    
    private func calculatePreview() -> (periods: Int, amountPerPeriod: Double)? {
        guard hasTargetDate else { return nil }
        
        let current = data.balance(for: account)
        let target = calculateAmount(targetAmountString)
        
        let today = Date()
        guard targetDate > today else { return nil }
        
        let periods = paymentFrequency.periodsBetween(from: today, to: targetDate)
        guard periods > 0 else { return nil }
        
        let remaining: Double
        
        switch account.accountType {
        case .goal:
            // Calculate savings needed
            remaining = max(target - current, 0)
        case .debt:
            // Calculate payoff amount
            remaining = max(current, 0)
        default:
            return nil
        }
        
        let amountPerPeriod = remaining / Double(periods)
        return (periods, amountPerPeriod)
    }
    
    // MARK: - Actions
    
    private func saveChanges() {
        let amount = calculateAmount(initialAmountString)
        let targetRaw = calculateAmount(targetAmountString)
        
        // Parse target: empty string or "0" → nil for debt accounts
        let target: Double?
        if isTargetEmptyOrZero(targetAmountString, val: targetRaw) {
            target = nil
        } else {
            target = targetRaw
        }
        
        data.updateAccount(
            id: account.id,
            name: name,
            initialAmount: amount,
            targetAmount: target,
            targetDate: hasTargetDate ? targetDate : nil,
            paymentFrequency: hasTargetDate ? paymentFrequency : nil
        )
        dismiss()
    }
    
    /// Helper to determine if we should treat the target as nil
    private func isTargetEmptyOrZero(_ str: String, val: Double) -> Bool {
        if str.isEmpty { return true }
        if !account.isCredit && val == 0 { return true } // Debit/Goal: 0 target = nil
        // For Credit/Debt: explicit 0 is a valid target (payoff goal), so we keep it unless empty
        return false
    }
}

        // 1. Determine which list to start with
        let templatesToFilter: [TransactionTemplate]
        
        if searchText.isEmpty {
            templatesToFilter = data.templates
        } else {
            templatesToFilter = data.templates.filter { template in
                let nameMatch = template.name.localizedCaseInsensitiveContains(searchText)
                let accName = accountName(for: template.accountID) ?? ""
                let accMatch = accName.localizedCaseInsensitiveContains(searchText)
                let catName = categoryName(for: template.categoryID) ?? ""
                let catMatch = catName.localizedCaseInsensitiveContains(searchText)
                return nameMatch || accMatch || catMatch
            }
        }
        
        // 2. Sort the resulting list Alphabetically
        return templatesToFilter.sorted {
            $0.name.localizedCaseInsensitiveCompare($1.name) == .orderedAscending
        }
    }
    
    // Split into groups
    var scheduledTemplates: [TransactionTemplate] {
        filteredTemplates.filter { $0.reminderSchedule != nil }

            let components = Calendar.current.dateComponents([.hour, .minute], from: reminderTime)
            let schedule = ReminderSchedule(
                frequency: frequency,
                time: components,
                dayOfWeek: frequency == .weekly ? dayOfWeek : nil,
                dayOfMonth: frequency == .monthly ? dayOfMonth : nil
            )
            template.reminderSchedule = schedule
        } else {
            template.reminderSchedule = nil
        }
        
        data.updateTemplate(template)
        dismiss()
    }
}

import SwiftUI

struct CategoryDetailView: View {
    @EnvironmentObject var data: AppData
    @AppStorage("currencySymbol") private var currencySymbol: String = "$"
    
    let category: Category
    let monthContext: Date // The month selected in Insights
    
    var body: some View {
        List {
            // Header: Total for this month
            Section {
                VStack(alignment: .center, spacing: 8) {
                    Text(category.emoji)
                        .font(.system(size: 50))
                    
                    Text(category.name)
                        .font(.headline)
                    
                    Text(totalAmount.formattedCurrency(symbol: currencySymbol))
                        .font(.system(size: 36, weight: .bold, design: .rounded))
                        .foregroundColor(category.type == .inflow ? .green : .red)
                    
                    Text(monthString)
                        .foregroundColor(.secondary)
                }
                .frame(maxWidth: .infinity)
                .padding(.vertical)
                .listRowBackground(Color.clear)
            }
            
            // Transaction List
            Section(header: Text("Transactions")) {
                if transactions.isEmpty {
                    Text("No transactions found.")
                        .foregroundColor(.secondary)
                } else {
                    ForEach(transactions) { transaction in
                        NavigationLink(destination: TransactionDetailView(transaction: transaction)) {
                            TransactionRow(transaction: transaction)
                        }
                    }
                }
            }
        }
        .navigationTitle(category.name)
        .navigationBarTitleDisplayMode(.inline)
    }
    
    // MARK: - Logic
    
    private var transactions: [Transaction] {
        let calendar = Calendar.current
        return data.transactions
            .filter { $0.categoryID == category.id }
            .filter { calendar.isDate($0.date, equalTo: monthContext, toGranularity: .month) }
            .sorted { $0.date > $1.date }
    }
    
    private var totalAmount: Double {
        transactions.reduce(0) { $0 + $1.amount }
    }
    
    private var monthString: String {
        let formatter = DateFormatter()
        formatter.dateFormat = "MMMM yyyy"
        return formatter.string(from: monthContext)
    }
}

import SwiftUI

/// Form to add a new category. Users choose a name, a type (inflow or outflow)
/// and optionally provide an emoji for quick visual identification.
struct AddCategoryView: View {
    @EnvironmentObject var data: AppData
    @Environment(\.dismiss) var dismiss
    @State private var name: String = ""
    @State private var type: CategoryType = .outflow
    @State private var emoji: String = ""
    
    // MARK: - Focus State
    enum Field: Hashable {
        case name
        case emoji
    }
    
    @FocusState private var focusedField: Field?

    var body: some View {
        NavigationStack {
            Form {
                Section(header: Text("Name")) {
                    TextField("Category Name", text: $name)
                        .focused($focusedField, equals: .name)
                        .onSubmit {
                            // When user taps Return on Name, move to Type picker or Emoji
                            // Since Type is a Picker (not text field), skip to Emoji
                            focusedField = .emoji
                        }
                }
                Section(header: Text("Type")) {
                    Picker("Type", selection: $type) {
                        ForEach(CategoryType.allCases) { t in
                            Text(t.rawValue.capitalized).tag(t)
                        }
                    }
                    .pickerStyle(SegmentedPickerStyle())
                }
                Section(header: Text("Emoji")) {
                    TextField("Emoji (Optional)", text: $emoji)
                        .focused($focusedField, equals: .emoji)
                        .onSubmit {
                            // When user taps Return on Emoji, attempt to save
                            if canSave {
                                saveCategory()
                            }
                        }
                }
            }
            .formToolbar(
                title: "New Category",
                canSave: canSave,
                onSave: {
                    saveCategory()
                },
                onCancel: {
                    dismiss()
                }
            )
            .onAppear {
                // Auto-focus on Name field when view appears
                focusedField = .name
            }
        }
    }
    
    // MARK: - Helpers
    
    private var canSave: Bool {
        !name.trimmingCharacters(in: .whitespaces).isEmpty
    }
    
    private func saveCategory() {
        data.addCategory(name: name, type: type, emoji: emoji)
        dismiss()
    }
}

import SwiftUI

struct ManageCategoriesView: View {
    @EnvironmentObject var data: AppData
    @State private var showingAddSheet = false
    @State private var categoryToEdit: Category?
    @State private var searchText = ""
    
    // SPLIT BY TYPE + SORTED ALPHABETICALLY
    var inflowCategories: [Category] {
        let list = searchText.isEmpty
            ? data.categories.filter { $0.type == .inflow }
            : data.categories.filter { $0.type == .inflow && $0.name.localizedCaseInsensitiveContains(searchText) }
        return list.sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }
    
    var outflowCategories: [Category] {
        let list = searchText.isEmpty
            ? data.categories.filter { $0.type == .outflow }
            : data.categories.filter { $0.type == .outflow && $0.name.localizedCaseInsensitiveContains(searchText) }
        return list.sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }
    
    var body: some View {
        ZStack {
            List {
                // INFLOW SECTION
                if !inflowCategories.isEmpty {
                    Section(header: Text("Inflow")) {
                        ForEach(inflowCategories) { category in
                            categoryRow(category)
                        }
                    }
                }
                
                // OUTFLOW SECTION
                if !outflowCategories.isEmpty {
                    Section(header: Text("Outflow")) {
                        ForEach(outflowCategories) { category in
                            categoryRow(category)
                        }
                    }
                }
                
                // EMPTY STATE (when search returns no results)
                if inflowCategories.isEmpty && outflowCategories.isEmpty {
                    Section {
                        Text("No categories found")
                            .foregroundColor(.secondary)
                            .frame(maxWidth: .infinity, alignment: .center)
                            .padding()
                    }
                }
            }
            .searchable(text: $searchText, placement: .navigationBarDrawer(displayMode: .always))
            .navigationTitle("Categories")
            // REMOVED: Toolbar button
            .sheet(isPresented: $showingAddSheet) {
                AddCategoryView().environmentObject(data)
            }
            .sheet(item: $categoryToEdit) { category in
                EditCategoryView(category: category).environmentObject(data)
            }
            
            // ADDED: FAB for adding categories (bottom-right)
            .floatingAddButton {
                showingAddSheet = true
            }
        }
    }
    
    // MARK: - Reusable Category Row
    
    @ViewBuilder
    private func categoryRow(_ category: Category) -> some View {
        HStack {
            Text(category.emoji)
            Text(category.name).foregroundColor(.primary)
            Spacer()
            // Badge removed since section headers now indicate type
        }
        .contentShape(Rectangle())
        .onTapGesture { categoryToEdit = category }
        .swipeActions(edge: .leading) {
            Button { categoryToEdit = category } label: {
                Label("Edit", systemImage: "pencil")
            }
            .tint(.blue)
        }
        .swipeActions(edge: .trailing) {
            Button(role: .destructive) { deleteCategory(category) } label: {
                Label("Delete", systemImage: "trash")
            }
        }
    }
    
    // MARK: - Actions
    
    private func deleteCategory(_ category: Category) {
        if let index = data.categories.firstIndex(where: { $0.id == category.id }) {
            data.categories.remove(at: index)
            data.save()
        }
    }
}
